function main(): void {
    let i: i32 = 0;
    loop {
        i += 1;
        if (i < 5) { continue; }
        break;
    }
    const a = loop { break 10; };
    const t = loop { break "x"; };
    const nested = loop {
        if (true) { break 1; }
        break 0;
    };
    console.log(a, t, nested);
}
